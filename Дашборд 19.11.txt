    <script>
        // --- НАСТРОЙКИ ---
        const SPREADSHEET_ID = '1aE1hYXgRzbEj7CwelMHCkOTkWc_A8p7uGE77xIJMTts';
        const SHEET_NAME = 'Лист1'; // Имя вкладки в таблице
        
        // --- ЛОГИКА ЗАГРУЗКИ ---
        async function loadData() {
            const statusBar = document.getElementById('status-bar');
            statusBar.style.display = 'none';

            try {
                // Формируем URL для получения JSON из Google Sheets (Google Visualization API)
                const url = `https://spreadsheets.google.com/feeds/list/${SPREADSHEET_ID}/${SHEET_NAME}/public/values?alt=json`;
                
                const response = await fetch(url);
                
                if (!response.ok) {
                    throw new Error('Таблица недоступна. Проверьте доступ "Все, у кого есть ссылка" -> Читатель.');
                }

                const data = await response.json();
                processData(data.feed.entry);

            } catch (error) {
                console.error(error);
                statusBar.style.display = 'block';
                statusBar.innerText = 'Ошибка: ' + error.message;
            }
        }

        // --- ОБРАБОТКА ДАННЫХ ---
        function processData(entries) {
            // Здесь должна быть логика парсинга ваших колонок.
            // Так как я не вижу названия колонок в вашей таблице, я оставлю заглушки.
            // Вам нужно будет сопоставить названия колонок из таблицы с переменными ниже.
            
            // ПРИМЕР: если в таблице колонка называется "План ВП", то в коде ищем entry.gsx$планвп.$t
            // Google конвертирует заголовки в нижний регистр и убирает пробелы для ключей gsx\$...

            let totalPlan = 0;
            let totalFact = 0;
            let totalPdz = 0;
            let managers = ;

            entries.forEach(entry => {
                // ВАЖНО: Замените 'gsx\$план' и т.д. на реальные названия ваших колонок (в нижнем регистре, без пробелов)
                // Пример: если колонка "План ВП", ключ будет gsx$планвп.$t
                
                const plan = parseFloat(entry.gsx$план.$t) || 0; // <-- ИЗМЕНИТЬ
                const fact = parseFloat(entry.gsx$факт.$t) || 0; // <-- ИЗМЕНИТЬ
                const pdz = parseFloat(entry.gsx$пдз.$t) || 0;  // <-- ИЗМЕНИТЬ
                const manager = entry.gsx$менеджер.$t || 'Без имени'; // <-- ИЗМЕНИТЬ

                totalPlan += plan;
                totalFact += fact;
                totalPdz += pdz;

                managers.push({
                    name: manager,
                    plan: plan,
                    fact: fact,
                    pdz: pdz
                });
            });

            // Расчет процентов
            const percent = totalPlan > 0 ? ((totalFact / totalPlan) * 100).toFixed(1) : 0;
            
            // Обновление UI
            document.getElementById('plan-company-val').innerText = percent + '%';
            document.querySelectorAll('.stat-value').innerText = totalPlan.toLocaleString(); // План
            document.querySelectorAll('.stat-value').innerText = totalFact.toLocaleString(); // Факт
            document.querySelectorAll('.stat-value').innerText = totalPdz.toLocaleString(); // ПДЗ
            
            // Список менеджеров
            const listEl = document.getElementById('managers-list');
            listEl.innerHTML = managers.map(m => 
                `<div style="display:flex; justify-content:space-between; border-bottom:1px solid #334155; padding:5px 0;">
                    <span>\${m.name}</span>
                    <span>${m.fact.toLocaleString()} / ${m.plan.toLocaleString()}</span>
                 </div>`
            ).join('');

            // Заглушки для графиков (нужно инициализировать Chart.js)
            initChart('chart-company', ['План', 'Факт'], [totalPlan, totalFact], ['#f59e0b', '#10b981']);
            initChart('chart-percent', ['Факт', 'Прогноз'], [percent, 105], ['#6366f1', '#8b5cf6']);
            initChart('chart-pdz', ['ПДЗ', 'Норма'], [totalPdz, 1000000], ['#ef4444', '#334155']);

            // Расчет рабочих дней (упрощенно: берем текущую дату)
            const today = new Date();
            const daysInMonth = new Date(today.getFullYear(), today.getMonth() + 1, 0).getDate();
            const workDays = Math.round(daysInMonth * 0.7); // ~70% рабочих дней
            document.getElementById('days-work').innerText = workDays;
            document.getElementById('days-left').innerText = workDays - today.getDate();
            document.getElementById('norm-today').innerText = (totalPlan / workDays).toLocaleString();
        }

        // --- ФУНКЦИЯ СОЗДАНИЯ ГРАФИКА ---
        function initChart(containerId, labels, dataValues, colors) {
            const ctx = document.getElementById(containerId).getContext('2d');
            new Chart(ctx, {
                type: 'bar',
                data: {
                    labels: labels,
                    datasets: [{
                        label: 'Значения',
                        data: dataValues,
                        backgroundColor: colors,
                        borderRadius: 5,
                        borderWidth: 0
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: { legend: { display: false } },
                    scales: { y: { beginAtZero: true } }
                }
            });
        }

        // Запуск
        loadData();
